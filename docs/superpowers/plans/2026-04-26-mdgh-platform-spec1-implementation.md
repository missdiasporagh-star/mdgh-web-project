# MDGH Platform — Spec 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the multi-page Miss Diaspora Ghana redesign at `staging.missdiasporagh.org` — a Neo-African Futurism Astro 5 platform with 6 wow features, Sanity CMS schemas wired in, a provider-agnostic payment abstraction, and a one-DNS-change cutover plan.

**Architecture:** New repo `mdgh-platform/` (sibling to `mdgh-web-project/`). Astro 5 SSR on Cloudflare Pages. React islands only on `/diaspora`, `/quiz`, `/heritage`. Tailwind v4 with CSS custom properties as design tokens. Self-hosted Fraunces + Inter + JetBrains Mono. Sanity Studio (Sanity-hosted at `mdgh.sanity.studio`) for content. Two Cloudflare Workers (Drive Importer + OG Image generator). Two Sanity datasets: `staging` and `production`.

**Tech Stack:** Astro 5 · TypeScript strict · Tailwind v4 · React 19 (islands only) · GSAP + ScrollTrigger · Lenis · `@sanity/client` · `@sanity/image-url` · `react-globe.gl` · `lite-youtube-embed` · Satori · Cloudflare Pages + Workers + R2 + KV · Vitest · Playwright · `@fontsource-variable/fraunces`, `@fontsource-variable/inter`, `@fontsource/jetbrains-mono`.

**File responsibility principles:**
- One Astro page = one route. Pages stay thin — they fetch data + compose components.
- One component = one visual responsibility.
- React islands live in `.tsx` files; static markup in `.astro`. The file extension itself signals client-side cost.
- All pure logic (payment interface, quiz scoring, Sanity queries) lives in `lib/` and is unit-tested.
- Sanity schemas are split by category (singleton vs collection vs object) so adding a content type is a single-file change.

```typescript
  projectId: process.env.SANITY_STUDIO_PROJECT_ID!,
  dataset,
  basePath: `/${dataset}`,
  plugins: [
    structureTool({
      structure: (S: any) =>
        S.list().title('Content').items([
          ...SINGLETONS.filter((s) => schemaTypes.find((t: any) => t.name === s)).map((id) =>
            S.listItem().title(SINGLETON_LABELS[id]).id(id).child(S.document().schemaType(id).documentId(id))
          ),
          S.divider(),

  document: {

        ? input.filter(({ action }: any) => !['delete', 'duplicate', 'unpublish'].includes(action ?? ''))
```

```tsx
import { useEffect, useState } from 'react';

const chapters = [
  { id: 'hero', label: 'Hero' },
  { id: 'mission', label: 'Mission' },
  { id: 'crown', label: 'The Crown' },
  { id: 'diaspora', label: 'Diaspora' },
  { id: 'cycle', label: 'Cycle' },
  { id: 'become', label: 'Become Her' },
];

export default function ChapterRail() {
  const [activeIdx, setActiveIdx] = useState(0);

  useEffect(() => {
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
    const sections = chapters.map((c) => document.querySelector<HTMLElement>(`[data-chapter="${c.id}"]`));
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((e) => {
          if (e.isIntersecting) {
            const idx = sections.findIndex((s) => s === e.target);
            if (idx >= 0) setActiveIdx(idx);
          }
        });
      },
      { threshold: 0.5 }
    );
    sections.forEach((s) => s && observer.observe(s));
    return () => observer.disconnect();
  }, []);

  return (
    <nav
      aria-label="Page chapters"
      style={{
        position: 'fixed',
        right: 24,
        top: '50%',
        transform: 'translateY(-50%)',
        zIndex: 30,
        display: 'flex',
        flexDirection: 'column',
        gap: 12,
      }}
    >
      {chapters.map((c, i) => (
        <a
          key={c.id}
          href={`#${c.id}`}
          aria-label={`Jump to ${c.label}`}
          style={{
            width: i === activeIdx ? 32 : 16,
            height: 4,
            borderRadius: 2,
            background: i === activeIdx ? 'var(--color-saffron)' : 'rgba(255,255,255,0.2)',
            transition: 'all var(--dur-base) var(--ease-emp)',
            display: 'block',
          }}
        />
      ))}
    </nav>
  );
}
```

```typescript
const DRIVE_RE = /\/d\/([a-zA-Z0-9_-]+)|id=([a-zA-Z0-9_-]+)/;

export function extractDriveFileId(url: string): string | null {
  const m = url.match(DRIVE_RE);
  return m?.[1] ?? m?.[2] ?? null;
}

export async function getServiceAccountToken(serviceAccountJson: string): Promise<string> {
  const sa = JSON.parse(serviceAccountJson);
  const now = Math.floor(Date.now() / 1000);
  const claim = {
    iss: sa.client_email,
    scope: 'https://www.googleapis.com/auth/drive.readonly',
    aud: 'https://oauth2.googleapis.com/token',
    exp: now + 3600,
    iat: now,
  };
  const header = { alg: 'RS256', typ: 'JWT' };

  const enc = (o: any) => btoa(JSON.stringify(o)).replace(/=/g, '').replace(/\+/g, '-').replace(/\//g, '_');
  const unsigned = `${enc(header)}.${enc(claim)}`;

  const pem = sa.private_key.replace(/-----BEGIN PRIVATE KEY-----|-----END PRIVATE KEY-----|\n/g, '');
  const der = Uint8Array.from(atob(pem), (c) => c.charCodeAt(0));
  const key = await crypto.subtle.importKey('pkcs8', der, { name: 'RSASSA-PKCS1-v1_5', hash: 'SHA-256' }, false, ['sign']);
  const sig = await crypto.subtle.sign({ name: 'RSASSA-PKCS1-v1_5' }, key, new TextEncoder().encode(unsigned));
  const sigB64 = btoa(String.fromCharCode(...new Uint8Array(sig))).replace(/=/g, '').replace(/\+/g, '-').replace(/\//g, '_');
  const jwt = `${unsigned}.${sigB64}`;

  const resp = await fetch('https://oauth2.googleapis.com/token', {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: `grant_type=urn:ietf:params:oauth:grant-type:jwt-bearer&assertion=${jwt}`,
  });
  if (!resp.ok) throw new Error(`Drive token failed: ${await resp.text()}`);
  const json = (await resp.json()) as { access_token: string };
  return json.access_token;
}

export async function fetchDriveFile(fileId: string, token: string): Promise<{ stream: ReadableStream; meta: { name: string; mimeType: string; size: number } }> {
  const metaResp = await fetch(`https://www.googleapis.com/drive/v3/files/${fileId}?fields=name,mimeType,size`, {
    headers: { Authorization: `Bearer ${token}` },
  });
  if (!metaResp.ok) throw new Error(`Drive meta failed: ${metaResp.status}`);
  const meta = (await metaResp.json()) as { name: string; mimeType: string; size: string };

  const dl = await fetch(`https://www.googleapis.com/drive/v3/files/${fileId}?alt=media`, {
    headers: { Authorization: `Bearer ${token}` },
  });
  if (!dl.ok || !dl.body) throw new Error(`Drive download failed: ${dl.status}`);

  return { stream: dl.body, meta: { name: meta.name, mimeType: meta.mimeType, size: Number(meta.size) } };
}
```

```typescript
import { extractDriveFileId, getServiceAccountToken, fetchDriveFile } from './drive';
import { uploadToR2, uploadToSanity } from './upload';

export interface Env {
  MEDIA: R2Bucket;
  DRIVE_SERVICE_ACCOUNT_JSON: string;
  SANITY_PROJECT_ID: string;
  SANITY_WRITE_TOKEN: string;
  ALLOWED_ORIGINS: string;
}

const VIDEO_R2_THRESHOLD = 5 * 1024 * 1024; // 5MB

function corsHeaders(origin: string | null, allowed: string): HeadersInit {
  const list = allowed.split(',').map((s) => s.trim());
  const ok = origin && list.includes(origin);
  return {
    'Access-Control-Allow-Origin': ok ? origin! : list[0]!,
    'Access-Control-Allow-Methods': 'POST, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type, Authorization',
    'Access-Control-Max-Age': '86400',
  };
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const origin = request.headers.get('Origin');
    if (request.method === 'OPTIONS') {
      return new Response(null, { status: 204, headers: corsHeaders(origin, env.ALLOWED_ORIGINS) });
    }
    if (request.method !== 'POST') return new Response('Method not allowed', { status: 405 });

    try {
      const body = (await request.json()) as { driveUrl: string; dataset: 'staging' | 'production' };
      const fileId = extractDriveFileId(body.driveUrl);
      if (!fileId) return json({ error: 'Could not extract Drive file ID' }, 400, origin, env);

      const token = await getServiceAccountToken(env.DRIVE_SERVICE_ACCOUNT_JSON);
      const { stream, meta } = await fetchDriveFile(fileId, token);

      const isVideo = meta.mimeType.startsWith('video/');
      const useR2 = isVideo && meta.size > VIDEO_R2_THRESHOLD;

      if (useR2) {
        const key = `videos/${fileId}-${meta.name}`;
        const url = await uploadToR2(env, key, stream, meta.mimeType);
        return json({ kind: 'video', r2VideoUrl: url, filename: meta.name }, 200, origin, env);
      }

      // For images (and small videos) upload to Sanity
      const buf = await new Response(stream).arrayBuffer();
      const result = await uploadToSanity({
        projectId: env.SANITY_PROJECT_ID,
        dataset: body.dataset,
        token: env.SANITY_WRITE_TOKEN,
        filename: meta.name,
        body: buf,
        contentType: meta.mimeType,
      });
      return json({ kind: 'image', sanityAssetId: result.assetId, url: result.url, filename: meta.name }, 200, origin, env);
    } catch (e: any) {
      return json({ error: e?.message ?? 'Unknown error' }, 500, origin, env);
    }
  },
};

function json(data: any, status: number, origin: string | null, env: Env): Response {
  return new Response(JSON.stringify(data), {
    status,
    headers: { 'Content-Type': 'application/json', ...corsHeaders(origin, env.ALLOWED_ORIGINS) },
  });
}
```

```tsx
import { useState } from 'react';
import { set, unset } from 'sanity';
import { Stack, TextInput, Button, Card, Text, Badge, Flex } from '@sanity/ui';
import { importFromDrive } from './client';

export function DriveAssetInput(props: any) {
  const { value, onChange, schemaType } = props;
  const [busy, setBusy] = useState(false);
  const [err, setErr] = useState<string | null>(null);
  const dataset = (typeof window !== 'undefined' && window.location.pathname.includes('/production') ? 'production' : 'staging') as 'staging' | 'production';

  async function onImport(driveUrl: string) {
    setBusy(true); setErr(null);
    const result = await importFromDrive(driveUrl, dataset);
    setBusy(false);
    if ('error' in result) { setErr(result.error); return; }
    if (result.kind === 'image') {
      onChange(set({ ...(value ?? {}), driveUrl, kind: 'image', sanityImage: { _type: 'image', asset: { _ref: result.sanityAssetId, _type: 'reference' } }, importStatus: 'imported' }));
    } else {
      onChange(set({ ...(value ?? {}), driveUrl, kind: 'video', r2VideoUrl: result.r2VideoUrl, importStatus: 'imported' }));
    }
  }

  return (
    <Stack space={3}>
      <Text size={1} weight="semibold">{schemaType.title}</Text>
      <TextInput
        placeholder="Paste a Google Drive share URL"
        defaultValue={value?.driveUrl ?? ''}
        onBlur={(e) => { const v = (e.target as HTMLInputElement).value; if (v && v !== value?.driveUrl) onImport(v); }}
        disabled={busy}
      />
      {busy && <Text size={1} muted>Importing… (videos can take a moment)</Text>}
      {err && <Card padding={3} tone="critical"><Text size={1}>{err}</Text></Card>}
      {value?.importStatus === 'imported' && (

          </Flex>
        </Card>
      )}
      {value && (
        <Button mode="ghost" tone="critical" text="Remove" onClick={() => onChange(unset())} />
      )}
    </Stack>
  );
}
```

```tsx
export default function HeritageTimeline({ queens, imageUrls }: { queens: QueenFull[]; imageUrls: Record<string, string> }) {
  const [active, setActive] = useState<QueenFull | null>(null);
  const railRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    function onKey(e: KeyboardEvent) {
      if (!railRef.current) return;
      const w = 380;
      if (e.key === 'ArrowRight') railRef.current.scrollBy({ left: w, behavior: 'smooth' });
      if (e.key === 'ArrowLeft') railRef.current.scrollBy({ left: -w, behavior: 'smooth' });
    }
    document.addEventListener('keydown', onKey);
    return () => document.removeEventListener('keydown', onKey);
  }, []);

  return (
    <>
      <div ref={railRef} style={{ display: 'flex', gap: 20, overflowX: 'auto', scrollSnapType: 'x mandatory', padding: '24px 24px 48px', scrollbarWidth: 'thin' }}>
        {queens.map((q) => <QueenCard key={q._id} queen={q} imageUrl={imageUrls[q._id]} onOpen={setActive} />)}
      </div>
      {active && <QueenModal queen={active} imageUrl={imageUrls[active._id]} onClose={() => setActive(null)} />}
    </>
  );
```

```typescript
export type Region = 'Greater Accra' | 'Ashanti' | 'Volta' | 'Northern' | 'Western' | 'Central' | 'Eastern';

export interface Weight { region: Region | string; weight: number; }
export interface AnsweredQuestion { weights: Weight[]; }
export interface QuizResult { winner: string | null; scores: Record<string, number>; }

export function scoreQuiz(answered: AnsweredQuestion[]): QuizResult {
  const scores: Record<string, number> = {};
  for (const a of answered) {
    for (const w of a.weights) {
      if (w.weight <= 0) continue;
      scores[w.region] = (scores[w.region] ?? 0) + w.weight;
    }
  }
  const entries = Object.entries(scores);
  if (entries.length === 0) return { winner: null, scores };
  // Highest score wins; ties resolve by alphabetical region name
  entries.sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]));
  return { winner: entries[0]![0], scores };
}

export function archetypeSlug(region: string): string {
  return region.toLowerCase().replace(/\s+/g, '-');
}
```

```tsx
export default function HeritageQuiz({ questions }: { questions: QuizQuestionDoc[] }) {
  const [answers, setAnswers] = useState<{ region: string; weight: number }[][]>([]);
  const idx = answers.length;

  function handleAnswer(optionIdx: number) {
    const q = questions[idx]!;
    const next = [...answers, q.options[optionIdx]!.weights];

    }
  }
```

```typescript
  const resp = await fetch(url);

  async fetch(req: Request): Promise<Response> {
    const url = new URL(req.url);
    if (!url.pathname.startsWith('/quiz')) return new Response('Not found', { status: 404 });
```

```typescript
import type { APIRoute } from 'astro';
import { createClient } from '@sanity/client';
import { env } from '~/lib/env';

export const prerender = false;

const writeClient = createClient({
  projectId: env.SANITY_PROJECT_ID,
  dataset: env.SANITY_DATASET,
  apiVersion: env.SANITY_API_VERSION,
  token: import.meta.env.SANITY_WRITE_TOKEN ?? env.SANITY_TOKEN,
  useCdn: false,
});

export const POST: APIRoute = async ({ request }) => {
  try {
    const form = await request.formData();
    const name = String(form.get('name') ?? '').trim();
    const email = String(form.get('email') ?? '').trim();
    const city = String(form.get('city') ?? '').trim();
    const notes = String(form.get('notes') ?? '').trim();

    if (!name || !email || !/^.+@.+\..+$/.test(email)) {
      return new Response(JSON.stringify({ error: 'Name and valid email are required.' }), {
        status: 400, headers: { 'Content-Type': 'application/json' },
      });
    }

    await writeClient.create({
      _type: 'waitlistEntry',
      name, email, city: city || undefined, notes: notes || undefined,
      sourceUrl: request.headers.get('Referer') ?? undefined,
      createdAt: new Date().toISOString(),
    });

    return new Response(JSON.stringify({ ok: true }), {
      status: 200, headers: { 'Content-Type': 'application/json' },
    });
  } catch (e: any) {
    return new Response(JSON.stringify({ error: e?.message ?? 'Unknown error' }), {
      status: 500, headers: { 'Content-Type': 'application/json' },
    });
  }
};
```
